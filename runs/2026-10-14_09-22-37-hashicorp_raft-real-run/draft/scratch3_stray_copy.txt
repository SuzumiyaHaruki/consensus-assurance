// Self-contained harness: only this file is copied into the captured package.

func resurrectEvent(t *testing.T, kind string, fields map[string]interface{}) {
	fields["event"] = kind
	b, err := json.Marshal(fields)
	if err != nil {
		t.Fatalf("marshal: %v", err)
	}
	fmt.Println("CA_EVENT " + string(b))
}

func resurrectServers(r *Raft) int {
	f := r.GetConfiguration()
	if err := f.Error(); err != nil {
		return -1
	}
	return len(f.Configuration().Servers)
}

func resurrectHasServer(r *Raft, id ServerID) bool {
	f := r.GetConfiguration()
	if err := f.Error(); err != nil {
		return false
	}
	for _, s := range f.Configuration().Servers {
		if s.ID == id {
			return true
		}
	}
	return false
}

func resurrectNumPeers(r *Raft) int {
	n, _ := strconv.Atoi(r.Stats()["num_peers"])
	return n
}

func resurrectLeaderAmong(rs []*Raft, d time.Duration) *Raft {
	deadline := time.Now().Add(d)
	for time.Now().Before(deadline) {
		for _, r := range rs {
			if r.State() == Leader {
				return r
			}
		}
		time.Sleep(5 * time.Millisecond)
	}
	return nil
}

// TestAssurance_RemovedVoterResurrectedByRestart removes a voter that is offline
// for its own removal, commits further entries, then restarts the leader with
// RestoreCommittedLogs enabled (and disabled as a control) and observes whether
// the restarted node still counts the removed voter.
func TestAssurance_RemovedVoterResurrectedByRestart(t *testing.T) {
	for _, mode := range []string{"restore_on", "restore_off"} {
		runResurrectCase(t, mode, mode == "restore_on")
	}
}

func runResurrectCase(t *testing.T, mode string, restore bool) {
	conf := inmemConfig(t)
	conf.RestoreCommittedLogs = restore
	conf.SnapshotThreshold = 1 << 40
	conf.SnapshotInterval = 10 * time.Minute
	conf.TrailingLogs = 10

	addrs := []ServerAddress{"A", "B", "C"}
	trans := make([]*InmemTransport, 3)
	confs := make([]*Config, 3)
	logs := make([]*InmemCommitTrackingStore, 3)
	stables := make([]*InmemStore, 3)
	snaps := make([]*InmemSnapshotStore, 3)
	for i, a := range addrs {
		_, trans[i] = NewInmemTransport(a)
		c := *conf
		c.LocalID = ServerID(a)
		confs[i] = &c
		logs[i] = NewInmemCommitTrackingStore()
		stables[i] = NewInmemStore()
		snaps[i] = NewInmemSnapshotStore()
	}
	for i := 0; i < 3; i++ {
		for j := 0; j < 3; j++ {
			if i != j {
				trans[i].Connect(addrs[j], trans[j])
			}
		}
	}
	cfg := Configuration{Servers: []Server{
		{Suffrage: Voter, ID: "A", Address: "A"},
		{Suffrage: Voter, ID: "B", Address: "B"},
		{Suffrage: Voter, ID: "C", Address: "C"},
	}}
	if err := BootstrapCluster(confs[0], logs[0], stables[0], snaps[0], trans[0], cfg); err != nil {
		t.Fatalf("[%s] bootstrap: %v", mode, err)
	}
	rafts := make([]*Raft, 3)
	for i := 0; i < 3; i++ {
		r, err := NewRaft(confs[i], new(MockFSM), logs[i], stables[i], snaps[i], trans[i])
		if err != nil {
			t.Fatalf("[%s] NewRaft %d: %v", mode, i, err)
		}
		rafts[i] = r
	}
	raftA, raftB, raftC := rafts[0], rafts[1], rafts[2]
	if resurrectLeaderAmong(rafts, 10*time.Second) != raftA {
		t.Fatalf("[%s] A did not become leader", mode)
	}
	for i := 0; i < 3; i++ {
		if err := raftA.Apply([]byte(fmt.Sprintf("cmd%d", i)), 2*time.Second).Error(); err != nil {
			t.Fatalf("[%s] apply %d: %v", mode, i, err)
		}
	}
	if err := raftA.Snapshot().Error(); err != nil {
		t.Fatalf("[%s] snapshot: %v", mode, err)
	}

	// C is offline for its own removal, so it never learns it was removed.
	if err := raftC.Shutdown().Error(); err != nil {
		t.Fatalf("[%s] shutdown C: %v", mode, err)
	}
	if err := raftA.RemoveServer("C", 0, 5*time.Second).Error(); err != nil {
		t.Fatalf("[%s] RemoveServer C: %v", mode, err)
	}
	// Push the persisted commit index past the removal entry.
	for i := 0; i < 2; i++ {
		if err := raftA.Apply([]byte(fmt.Sprintf("after%d", i)), 2*time.Second).Error(); err != nil {
			t.Fatalf("[%s] apply after %d: %v", mode, i, err)
		}
	}
	staged, err := logs[0].GetCommitIndex()
	if err != nil {
		t.Fatalf("[%s] GetCommitIndex: %v", mode, err)
	}
	resurrectEvent(t, "before_restart_"+mode, map[string]interface{}{
		"mode":              mode,
		"a_servers":         resurrectServers(raftA),
		"a_has_c":           resurrectHasServer(raftA, "C"),
		"b_servers":         resurrectServers(raftB),
		"b_has_c":           resurrectHasServer(raftB, "C"),
		"staged_commit_idx": staged,
		"a_log_last":        raftA.LastIndex(),
		"snapshots":         func() int { l, _ := snaps[0].List(); return len(l) }(),
	})

	if err := raftA.Shutdown().Error(); err != nil {
		t.Fatalf("[%s] shutdown A: %v", mode, err)
	}
	_, transA2 := NewInmemTransport("A")
	transA2.Connect("B", trans[1])
	trans[1].Connect("A", transA2)
	raftA, err = NewRaft(confs[0], new(MockFSM), logs[0], stables[0], snaps[0], transA2)
	if err != nil {
		t.Fatalf("[%s] restart A: %v", mode, err)
	}
	resurrectEvent(t, "after_restart_"+mode, map[string]interface{}{
		"mode":        mode,
		"a_servers":   resurrectServers(raftA),
		"a_has_c":     resurrectHasServer(raftA, "C"),
		"a_num_peers": resurrectNumPeers(raftA),
		"b_servers":   resurrectServers(raftB),
		"b_has_c":     resurrectHasServer(raftB, "C"),
		"b_num_peers": resurrectNumPeers(raftB),
		"staged_commit": staged,
	})
	_ = raftA.Shutdown().Error()
	_ = raftB.Shutdown().Error()
}
